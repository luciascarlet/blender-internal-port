cmake_minimum_required(VERSION 3.16)
project(blender_internal_port LANGUAGES CXX)
add_library(blender_internal_core SHARED native/classic.cpp)
target_compile_features(blender_internal_core PRIVATE cxx_std_17)
set_target_properties(blender_internal_core PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
  target_compile_options(blender_internal_core PRIVATE /W4)
else()
  target_compile_options(blender_internal_core PRIVATE -Wall -Wextra -Wno-unused-parameter -ffp-contract=off)
endif()
install(TARGETS blender_internal_core LIBRARY DESTINATION blender_internal RUNTIME DESTINATION blender_internal)
